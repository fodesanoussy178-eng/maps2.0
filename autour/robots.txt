# Les pages publiques sont ouvertes à l'indexation : l'accueil et les deux

User-agent: *
Allow: /

# LES ROUTES DE DONNÉES NE SONT PAS DES PAGES.
#
# `/api/*` rend du JSON : rien d'indexable, et surtout rien de gratuit. Chaque
# appel non servi par le cache déclenche une requête vers Overpass, Google
# Places, DATAtourisme ou un service public. Un robot qui les parcourt paie ce
# coût sans produire une seule ligne de résultat de recherche.
Disallow: /api/

# ajoutée le jour où le fichier existera réellement.
#
# Note pour les déploiements de prévisualisation Vercel : ce fichier est servi
# à l'identique sur `*.vercel.app`, qui autoriserait donc l'indexation d'un
